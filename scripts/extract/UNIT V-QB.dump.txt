bool dfs(int n, int u, int parent) {
    visited[u] = true;
    for (int v = 0; v < n; v++) {
        if (adj[u][v] == 1) {
            if (!visited[v]) {

#include <iostream>
#include <string>
using namespace std;
int adj[20][20];
int dist[20];
void addEdge(int u, int v) {
    adj[u][v] = 1;
    adj[v][u] = 1;
}
void bfs(int n, int source) {
    int queue[20];
    int front = 0;
    int rear = 0;
    for (int i = 0; i < n; i++) {
        dist[i] = -1;
    }
    dist[source] = 0;
    queue[rear] = source;
    rear = rear + 1;
    while (front < rear) {
        int u = queue[front];
        front = front + 1;
        for (int v = 0; v < n; v++) {
            if (adj[u][v] == 1 && dist[v] == -1) {
                dist[v] = dist[u] + 1;
                queue[rear] = v;
                rear = rear + 1;
            }
        }
    }
}

void dijkstra(int n, int source, int dist[]) {
    bool done[20];
    for (int i = 0; i < n; i++) {
        dist[i] = 1000000000;
        done[i] = false;
    }
    dist[source] = 0;
    for (int count = 0; count < n; count++) {
        int u = -1;
        for (int v = 0; v < n; v++) {
            if (!done[v] && dist[v] != 1000000000 && (u == -1 || dist[v] < dist[u])) {
                u = v;
            }
        }

            break;

            if (graph[u][v] > 0 && !done[v] && dist[u] + graph[u][v] < dist[v]) {
                dist[v] = dist[u] + graph[u][v];

    }
}

#include <iostream>
using namespace std;
int main() {
    int n, m;
    cin >> n >> m;
    int adj[20][20];
    int indeg[20];
    for (int i = 0; i < n; i++) {
        indeg[i] = 0;
        for (int j = 0; j < n; j++) {
            adj[i][j] = 0;
        }
    }
    for (int i = 0; i < m; i++) {
        int a, b;
        cin >> a >> b;
        if (adj[a][b] == 0) {
            adj[a][b] = 1;
            indeg[b] = indeg[b] + 1;
        }
    }
    bool done[20];
    for (int i = 0; i < n; i++) {
        done[i] = false;
    }
    int taken = 0;
    int semesters = 0;
    while (taken < n) {
        int batch[20];
        int count = 0;
        for (int i = 0; i < n; i++) {
            if (!done[i] && indeg[i] == 0) {
                batch[count] = i;
                count = count + 1;
            }
        }
        if (count == 0) {
            cout << "Impossible: Cycle in prerequisites" << endl;
            return 0;
        }
        for (int k = 0; k < count; k++) {
            int u = batch[k];
            done[u] = true;
            taken = taken + 1;
            for (int j = 0; j < n; j++) {
                if (adj[u][j] == 1) {
                    indeg[j] = indeg[j] - 1;
                }
            }
        }
        semesters = semesters + 1;
    }
    cout << "Semesters: " << semesters << endl;
    return 0;
}

                        rear = rear + 1;
                    } else if (color[v] == color[u]) {

                    }
                }
            }
        }
    }

    cout << "Team A:";
    for (int i = 0; i < n; i++) {
        if (color[i] == 0) {
            cout << " " << i;
        }

    for (int i = 0; i < n; i++) {
        if (color[i] == 1) {
            cout << " " << i;
        }
    }
    cout << endl;
    return 0;

Write a program that prints the minimum total cost of travelling from the city 0 to the city n-1. If the city n-1 cannot be reached from the city 0, print "No Route". If a loop with a negative total cost can be reached from the city 0, print "Negative Cycle" instead.

4. Problem Statement
Hemalatha works for a telecom company that has built phone lines between the villages of a district. There are n villages numbered from 0 to n-1, and every phone line joins two villages and works in both directions. Two villages are in the same group if they can talk to each other, directly or through other villages. A village with no phone line forms a group alone. The network is called connected if all the villages are in one group.
Write a program that prints the number of groups of villages, the number of villages in the largest group, and whether the network is connected.
Input Format 
The first line of input consists of two space-separated integers n and m representing the number of villages and the number of phone lines.
Each of the next m lines consists of two space-separated integers u and v representing a phone line between the village u and the village v.
Output Format 
The first line of output prints "Components: " followed by an integer representing the number of groups of villages.
The second line of output prints "Largest Component: " followed by an integer representing the number of villages in the largest group.
The third line of output prints "Connected: Yes" if all the villages are in one group, or "Connected: No" otherwise.
Refer to the sample output for the formatting specifications.
Constraints 
In this scenario, the test cases fall under the following constraints:
1 ≤ n ≤ 15
0 ≤ m ≤ min(50, n × (n − 1) / 2)
0 ≤ u, v < n
There are no self-loops and no repeated phone lines.
[[IMAGE IN PARAGRAPH]]
Solution
#include <iostream>
using namespace std;
int n;
int adj[20][20];
bool visited[20];
int dfs(int u) {
    visited[u] = true;
    int size = 1;
    for (int v = 0; v < n; v++) {
        if (adj[u][v] == 1 && !visited[v]) {
            size = size + dfs(v);
        }
    }
    return size;
}
int main() {
    int m;
    cin >> n >> m;
    for (int i = 0; i < m; i++) {
        int u, v;
        cin >> u >> v;
        adj[u][v] = 1;
        adj[v][u] = 1;
    }
    int components = 0;
    int largest = 0;
    for (int i = 0; i < n; i++) {
        if (!visited[i]) {
            int size = dfs(i);
            components = components + 1;
            if (size > largest) {
                largest = size;
            }
        }
    }
    cout << "Components: " << components << endl;
    cout << "Largest Component: " << largest << endl;
    if (components == 1) {
        cout << "Connected: Yes" << endl;
    } else {
        cout << "Connected: No" << endl;
    }
    return 0;
}
