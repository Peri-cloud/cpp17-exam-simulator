using namespace std;
class Complex {
    int real;
    int imag;
public:
    Complex(int r, int i) {
        real = r;
        imag = i;
    }
    Complex operator+(Complex other) {
        return Complex(real + other.real, imag + other.imag);
    }
    Complex operator*(Complex other) {
        return Complex(real * other.real - imag * other.imag, real * other.imag + imag * other.real);

    }
};

    cout << "Product: ";
    product.display();
    return 0;
}

Solution
#include <iostream>
#include <string>
using namespace std;
class Score {
    int value;
    int highest;
public:
    Score(int start) {
        value = start;
        highest = start;
    }
    Score& operator++() {
        value = value + 1;
        if (value > highest) {
            highest = value;
        }
        return *this;
    }
    Score& operator--() {
        if (value > 0) {
            value = value - 1;
        }
        return *this;
    }
    int getValue() {
        return value;
    }
    int getHighest() {
        return highest;
    }
};
int main() {
    int start;
    string commands;
    cin >> start >> commands;
    Score s(start);
    for (int i = 0; i < commands.length(); i++) {
        if (commands[i] == '+') {
            ++s;
        } else {
            --s;
        }
    }
    cout << "Final Score: " << s.getValue() << endl;
    cout << "Highest Score: " << s.getHighest() << endl;
    return 0;
}

#include <iostream>
#include <iomanip>
using namespace std;
class Payment {
public:
    virtual double fee(double amount) = 0;
    virtual void name() = 0;
    virtual ~Payment() {
        cout << "Payment Closed" << endl;
    }
};
class CardPayment : public Payment {
public:
    double fee(double amount) {
        return amount * 2.0 / 100;
    }
    void name() {
        cout << "Method: Card" << endl;
    }
    ~CardPayment() {
        cout << "Card Closed" << endl;
    }
};
class UpiPayment : public Payment {
public:
    double fee(double amount) {
        return amount * 0.5 / 100;
    }
    void name() {
        cout << "Method: UPI" << endl;
    }
    ~UpiPayment() {
        cout << "UPI Closed" << endl;
    }
};
int main() {
    int choice;
    double amount;
    cin >> choice >> amount;
    Payment* p;
    if (choice == 1) {
        p = new CardPayment();
    } else {
        p = new UpiPayment();
    }
    cout << fixed << setprecision(2);
    p->name();
    cout << "Fee: " << p->fee(amount) << endl;
    cout << "Total: " << amount + p->fee(amount) << endl;
    delete p;
    return 0;
}
