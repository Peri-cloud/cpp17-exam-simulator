The balance factor of a node is the height of its left subtree minus the height of its right subtree. If the balance factor of the root is greater than 1, the tree is rebalanced with a single right rotation at the root. If the balance factor of the root is less than -1, the tree is rebalanced with a single left rotation at the root. Otherwise, the tree is not changed. #include <iostream>
using namespace std;
struct Node {
    int data;
    Node* left;
    Node* right;
};
Node* insert(Node* root, int value) {
    if (root == NULL) {
        Node* node = new Node;
        node->data = value;
        node->left = NULL;
        node->right = NULL;
        return node;
    }
    if (value < root->data) {
        root->left = insert(root->left, value);
    } else {
        root->right = insert(root->right, value);
    }
    return root;
}
int height(Node* root) {
    if (root == NULL) {
        return 0;
    }
    int l = height(root->left);
    int r = height(root->right);
    if (l > r) {
        return l + 1;
    }
    return r + 1;
}
int balanceFactor(Node* root) {
    return height(root->left) - height(root->right);
}
Node* rotateRight(Node* y) {
    Node* x = y->left;
    y->left = x->right;
    x->right = y;
    return x;
}
Node* rotateLeft(Node* x) {
    Node* y = x->right;
    x->right = y->left;
    y->left = x;
    return y;
}
Node* rebalance(Node* root) {
    int bf = balanceFactor(root);
    if (bf > 1) {
        return rotateRight(root);
    }
    if (bf < -1) {
        return rotateLeft(root);
    }
    return root;
}
void preorder(Node* root) {
    if (root == NULL) {
        return;
    }
    cout << " " << root->data;
    preorder(root->left);
    preorder(root->right);
}
void printPreorder(Node* root) {
    cout << "Preorder After Balancing:";
    preorder(root);
    cout << endl;
}

#include <iostream>
#include <string>
using namespace std;
struct Node {
    char data;
    Node* left;
    Node* right;
};
Node* newNode(char c) {
    Node* node = new Node;
    node->data = c;
    node->left = NULL;
    node->right = NULL;
    return node;
}
Node* build(string postfix) {
    Node* stack[50];
    int top = -1;
    for (int i = 0; i < postfix.length(); i++) {
        char c = postfix[i];
        Node* node = newNode(c);
        if (c < '0' || c > '9') {
            Node* right = stack[top];
            top = top - 1;
            Node* left = stack[top];
            top = top - 1;
            node->left = left;
            node->right = right;
        }
        top = top + 1;
        stack[top] = node;
    }
    return stack[top];
}
int evaluate(Node* root) {
    if (root->left == NULL && root->right == NULL) {
        return root->data - '0';
    }
    int l = evaluate(root->left);
    int r = evaluate(root->right);
    if (root->data == '+') {
        return l + r;
    }
    if (root->data == '-') {
        return l - r;
    }
    if (root->data == '*') {
        return l * r;
    }
    return l / r;
}
void infix(Node* root) {
    if (root->left == NULL && root->right == NULL) {
        cout << root->data;
        return;
    }
    cout << "(";
    infix(root->left);
    cout << root->data;
    infix(root->right);
    cout << ")";
}

    cout << endl;

#include <iostream>
#include <string>
using namespace std;
struct TrieNode {
    TrieNode* children[26];
    bool isEnd;
};
TrieNode* createNode() {
    TrieNode* node = new TrieNode;
    for (int i = 0; i < 26; i++) {
        node->children[i] = NULL;
    }
    node->isEnd = false;
    return node;
}
class Trie {
    TrieNode* root;
public:
    Trie() {
        root = createNode();
    }
    void insert(string word) {
        TrieNode* temp = root;
        for (int i = 0; i < word.length(); i++) {
            int index = word[i] - 'a';
            if (temp->children[index] == NULL) {
                temp->children[index] = createNode();
            }
            temp = temp->children[index];
        }
        temp->isEnd = true;
    }
    string search(string word) {
        TrieNode* temp = root;
        for (int i = 0; i < word.length(); i++) {
            int index = word[i] - 'a';
            if (temp->children[index] == NULL) {
                return "Not Found";
            }
            temp = temp->children[index];
        }
        if (temp->isEnd) {
            return "Found";
        }
        return "Not Found";
    }
};

Q2.Problem Statement
Dr. Subramanian teaches data structures and checks the colouring of the trees drawn by his students. Every node of a tree is coloured either red (R) or black (B). The tree is a correct red-black tree colouring if it follows all these rules: the root is black; a red node never has a red child; and every path from the root down to an empty link has the same number of black nodes. 
The empty links are not counted as nodes. The number of black nodes on such a path is called the black height of the tree.The ordering of the keys is already correct, and it does not have to be checked.

Input Format 
The first line of input consists of an integer n representing the number of nodes in the tree.
Each of the next n lines consists of an integer key, a character color, and two integers l and r separated by spaces, representing the key of the node i, its colour (R or B), and the number of its left child and the number of its right child, where i is the position of the line starting from 0. The value -1 means that the child does not exist. The node 0 is the root.
Output Format 
If the tree follows all the rules, the first line of output prints "Valid Red-Black Tree", and the second line of output prints "Black Height: " followed by an integer representing the black height of the tree.
Otherwise, the first line of output prints "Invalid Red-Black Tree".

Constraints 
In this scenario, the test cases fall under the following constraints:

-1 ≤ l, r < n

#include <iostream>

int n;
char color[50];
int leftChild[50];
int rightChild[50];
int blackHeight(int node) {
    if (node == -1) {
        return 0;
    }
    if (color[node] == 'R') {
        if (leftChild[node] != -1 && color[leftChild[node]] == 'R') {
            return -1;
        }
        if (rightChild[node] != -1 && color[rightChild[node]] == 'R') {
            return -1;

    int r = blackHeight(rightChild[node]);
    if (l == -1 || r == -1 || l != r) {
        return -1;
    }
    if (color[node] == 'B') {
        return l + 1;

Q3.Problem Statement
Uma is a developer of a search box that suggests words while a user is typing. A dictionary of words is given, and for every prefix that the user types, the search box must show how many words of the dictionary start with that prefix. A word also starts with itself, so a word that is equal to the prefix is counted too.
Write a program that stores the dictionary in a trie, where every node keeps the number of the words that pass through it, and prints the number of words for every prefix typed by the user.
Input Format 
The first line of input consists of an integer n representing the number of words in the dictionary.
The second line of input consists of n space-separated strings representing the words of the dictionary.
The third line of input consists of an integer q representing the number of prefixes typed.
The fourth line of input consists of q space-separated strings representing the prefixes.
Output Format 
The output contains q lines.
Each line prints the prefix, ": ", and an integer representing the number of words of the dictionary that start with the prefix.
Refer to the sample output for the formatting specifications.
Constraints 
In this scenario, the test cases fall under the following constraints:
1 ≤ n, q ≤ 50
1 ≤ length of every word and every prefix ≤ 20, and they are made of lowercase letters only.
All the words of the dictionary are different.
Sample Input 
4
red rose rat ram
2
r ra
Sample Output 
r: 4
ra: 2
Sample Input 
3
sky ski sketch
3
sk ski skyline
Sample Output 
sk: 3
ski: 1
skyline: 0
Solution:-
#include <iostream>
#include <string>
using namespace std;
struct TrieNode {
    TrieNode* children[26];
    int count;
};
TrieNode* createNode() {
    TrieNode* node = new TrieNode;
    for (int i = 0; i < 26; i++) {
        node->children[i] = NULL;
    }
    node->count = 0;
    return node;
}
int main() {
    int n, q;
    cin >> n;
    TrieNode* root = createNode();
    for (int i = 0; i < n; i++) {
        string word;
        cin >> word;
        TrieNode* temp = root;
        for (int j = 0; j < word.length(); j++) {
            int index = word[j] - 'a';
            if (temp->children[index] == NULL) {
                temp->children[index] = createNode();
            }
            temp = temp->children[index];
            temp->count = temp->count + 1;
        }
    }
    cin >> q;
    for (int i = 0; i < q; i++) {
        string prefix;
        cin >> prefix;
        TrieNode* temp = root;
        int result = 0;
        bool found = true;
        for (int j = 0; j < prefix.length(); j++) {
            int index = prefix[j] - 'a';
            if (temp->children[index] == NULL) {
                found = false;
                break;
            }
            temp = temp->children[index];
        }
        if (found) {
            result = temp->count;
        }
        cout << prefix << ": " << result << endl;
    }
    return 0;
}
Q4.Problem Statement
Ms. Kalpana teaches AVL trees and wants to tell her students which rotation is needed at the root of a tree. The keys are added one by one, in the given order, to an empty binary search tree, where the keys smaller than the key of a node go to its left subtree, and the other keys go to its right subtree. No rotation is done while the keys are added.
The height of an empty tree is 0, and the balance factor of a node is the height of its left subtree minus the height of its right subtree. If the balance factor of the root is between -1 and 1, no rotation is needed. Otherwise, the rotation is decided by the balance factor of the taller child of the root:
Balance factor of the root greater than 1, and balance factor of the left child greater than or equal to 0: LL (single right rotation)
Balance factor of the root greater than 1, and balance factor of the left child less than 0: LR (double rotation, left and then right)
Balance factor of the root less than -1, and balance factor of the right child less than or equal to 0: RR (single left rotation)
Balance factor of the root less than -1, and balance factor of the right child greater than 0: RL (double rotation, right and then left)
Write a program that prints the balance factor of the root and the rotation needed.
Input Format 
The first line of input consists of an integer n representing the number of keys.
The second line of input consists of n space-separated integers representing the keys in the order in which they are added.
Output Format 
The first line of output prints "Balance Factor: " followed by an integer representing the balance factor of the root.
The second line of output prints "Rotation: " followed by LL, LR, RR, RL, or None.
Refer to the sample output for the formatting specifications.
Constraints 
In this scenario, the test cases fall under the following constraints:
1 ≤ n ≤ 20
1 ≤ key ≤ 1000, and all the keys are different.
Sample Input 
4
8 5 10 3
Sample Output 
Balance Factor: 1
Rotation: None
Sample Input 
6
60 40 80 30 50 20
Sample Output 
Balance Factor: 2
Rotation: LL
Solution:-
#include <iostream>
using namespace std;
int val[20];
int leftChild[20];
int rightChild[20];
int height(int node) {
    if (node == -1) {
        return 0;
    }
    int l = height(leftChild[node]);
    int r = height(rightChild[node]);
    if (l > r) {
        return l + 1;
    }
    return r + 1;
}
int balance(int node) {
    return height(leftChild[node]) - height(rightChild[node]);
}
int main() {
    int n;
    cin >> n;
    for (int i = 0; i < n; i++) {
        cin >> val[i];
        leftChild[i] = -1;
        rightChild[i] = -1;
        if (i > 0) {
            int cur = 0;
            while (true) {
                if (val[i] < val[cur]) {
                    if (leftChild[cur] == -1) {
                        leftChild[cur] = i;
                        break;
                    }
                    cur = leftChild[cur];
                } else {
                    if (rightChild[cur] == -1) {
                        rightChild[cur] = i;
                        break;
                    }
                    cur = rightChild[cur];
                }
            }
        }
    }
    int bf = balance(0);
    cout << "Balance Factor: " << bf << endl;
    if (bf > 1) {
        if (balance(leftChild[0]) >= 0) {
            cout << "Rotation: LL" << endl;
        } else {
            cout << "Rotation: LR" << endl;
        }
    } else if (bf < -1) {
        if (balance(rightChild[0]) <= 0) {
            cout << "Rotation: RR" << endl;
        } else {
            cout << "Rotation: RL" << endl;
        }
    } else {
        cout << "Rotation: None" << endl;
    }
    return 0;
}
