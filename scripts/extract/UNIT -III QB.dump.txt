#include <iostream>
using namespace std;
struct Node {
    int data;
    Node* next;
};
Node* insertEnd(Node* head, int value) {
    Node* newNode = new Node;
    newNode->data = value;
    newNode->next = NULL;
    if (head == NULL) {
        return newNode;
    }
    Node* temp = head;
    while (temp->next != NULL) {
        temp = temp->next;
    }
    temp->next = newNode;
    return head;
}
Node* reverseList(Node* head) {
    Node* prev = NULL;
    Node* curr = head;
    while (curr != NULL) {
        Node* nextNode = curr->next;
        curr->next = prev;
        prev = curr;
        curr = nextNode;
    }
    return prev;
}
void display(Node* head) {
    Node* temp = head;
    while (temp != NULL) {
        cout << temp->data;
        if (temp->next != NULL) {
            cout << " ";
        }
        temp = temp->next;
    }
    cout << endl;
}

Header
#include <iostream>
using namespace std;
class CircularQueue {
    int arr[100];
    int capacity;
    int front;
    int rear;
    int count;
public:
    CircularQueue(int c) {
        capacity = c;
        front = 0;
        rear = -1;
        count = 0;
    }
    void process(int type, int value) {
        if (type == 1) {
            if (count == capacity) {
                cout << "Queue Full" << endl;
                return;
            }
            rear = (rear + 1) % capacity;
            arr[rear] = value;
            count = count + 1;
            cout << "Enqueued " << value << endl;
        } else {
            if (count == 0) {
                cout << "Queue Empty" << endl;
                return;
            }
            cout << "Dequeued " << arr[front] << endl;
            front = (front + 1) % capacity;
            count = count - 1;
        }
    }
    void show() {
        if (count == 0) {
            cout << "Queue: Empty" << endl;
            return;
        }
        cout << "Queue:";
        for (int i = 0; i < count; i++) {
            cout << " " << arr[(front + i) % capacity];
        }
        cout << endl;
    }
};
Footer
int main() {
    int c, n;
    cin >> c >> n;
    CircularQueue q(c);
    for (int i = 0; i < n; i++) {
        int type, value;
        cin >> type >> value;
        q.process(type, value);
    }
    q.show();
    return 0;
}

#include <iostream>

struct Node {
    int data;
    Node* next;
};
Node* insertEnd(Node* head, int value) {
    Node* newNode = new Node;
    newNode->data = value;
    if (head == NULL) {
        newNode->next = newNode;
        return newNode;
    }
    Node* temp = head;
    while (temp->next != head) {
        temp = temp->next;

    for (int i = 0; i < steps; i++) {

#include <iostream>
#include <string>
using namespace std;
int evaluate(string expr) {
    int stack[50];
    int top = -1;
    for (int i = 0; i < expr.length(); i++) {
        char c = expr[i];
        if (c >= '0' && c <= '9') {
            top = top + 1;
            stack[top] = c - '0';
        } else {
            int b = stack[top];
            top = top - 1;
            int a = stack[top];
            top = top - 1;
            int result = 0;
            if (c == '+') {
                result = a + b;
            } else if (c == '-') {
                result = a - b;
            } else if (c == '*') {
                result = a * b;
            } else {
                result = a / b;
            }
            top = top + 1;
            stack[top] = result;
        }
    }
    return stack[top];
}

Problem Statement
Lavanya is teaching a compiler design lab and wants to show her students how a postfix expression turns into an expression tree. In an expression tree, every leaf holds an operand and every internal node holds an operator whose left and right children are its two operands.
Write a program that reads a postfix expression and builds its expression tree using a stack of tree nodes: push a new node for every operand, and for every operator create a node, make the top of the stack its right child, the next one its left child, and push it back. Finally, print the expression in infix form by an inorder traversal, wrapping every operator node and its two operands in a pair of parentheses. A single operand is printed without parentheses.
Note: Use the malloc() function to allocate memory for the nodes of the tree.
Input Format 
The first line of input consists of a string s representing the postfix expression with no spaces.
Output Format 
The first line of output prints "Infix Expression: " followed by the fully parenthesized infix form of the expression.
Refer to the sample output for the formatting specifications.
Constraints 
In this scenario, the test cases fall under the following constraints:
1 ≤ length of s ≤ 50
s is a valid postfix expression containing lowercase letters as operands and the operators + - * /

Solution
#include <stdio.h>
#include <stdlib.h>
struct Node {
    char data;
    struct Node *left;
    struct Node *right;
};
struct Node *newNode(char c) {
    struct Node *node = (struct Node *)malloc(sizeof(struct Node));
    node->data = c;
    node->left = NULL;
    node->right = NULL;
    return node;
}
void printInfix(struct Node *root) {
    if (root == NULL) {
        return;
    }
    if (root->left == NULL && root->right == NULL) {
        printf("%c", root->data);
        return;
    }
    printf("(");
    printInfix(root->left);
    printf("%c", root->data);
    printInfix(root->right);
    printf(")");
}
int main() {
    char s[55];
    struct Node *stack[55];
    int top = -1;
    scanf("%s", s);
    for (int i = 0; s[i] != '\0'; i++) {
        char c = s[i];
        struct Node *node = newNode(c);
        if (c < 'a' || c > 'z') {
            node->right = stack[top];
            top--;
            node->left = stack[top];
            top--;
        }
        top++;
        stack[top] = node;
    }
    printf("Infix Expression: ");
    printInfix(stack[top]);
    printf("\n");
    return 0;
}
Q4.
Problem Statement
Suresh organises the digital catalogue of a library in a binary search tree of book IDs. Each book ID is inserted in the order it was received: an ID smaller than the current node goes to the left subtree, and an ID larger than the current node goes to the right subtree. All book IDs in the catalogue are different.
Write a program that builds the binary search tree from the given IDs and then searches for a given key starting from the root. The depth of a node is the number of edges on the path from the root to that node, so the root has depth 0. If the key is in the tree, print its depth, otherwise report that it was not found.
Note: Use the malloc() function to allocate memory for the nodes of the tree.
Input Format 
The first line of input consists of an integer n representing the number of book IDs.
The second line of input consists of n space-separated integers representing the book IDs in the order they are inserted.
The third line of input consists of an integer key representing the book ID to search for.
Output Format 
If the key is present, the first line of output prints "Found at depth " followed by an integer representing the depth of the node holding the key.
If the key is not present, the first line of output prints the string "Not Found".

#include <stdio.h>
#include <stdlib.h>
struct Node {
    int data;
    struct Node *left;
    struct Node *right;
};
struct Node *insert(struct Node *root, int x) {
    if (root == NULL) {
        struct Node *node = (struct Node *)malloc(sizeof(struct Node));
        node->data = x;
        node->left = NULL;
        node->right = NULL;
        return node;
    }
    if (x < root->data) {
        root->left = insert(root->left, x);
    } else {
        root->right = insert(root->right, x);
    }
    return root;
}
int main() {
    int n, key;
    struct Node *root = NULL;
    scanf("%d", &n);
    for (int i = 0; i < n; i++) {
        int x;
        scanf("%d", &x);
        root = insert(root, x);
    }
    scanf("%d", &key);
    struct Node *curr = root;
    int depth = 0;
    while (curr != NULL && curr->data != key) {
        if (key < curr->data) {
            curr = curr->left;
        } else {
            curr = curr->right;
        }
        depth++;
    }
    if (curr == NULL) {
        printf("Not Found\n");
    } else {
        printf("Found at depth %d\n", depth);
    }
    return 0;
}
