Q1. Problem Statement
Dinesh is a mathematics teacher who prepares number puzzles for his students. A prime number is a number greater than 1 that has no divisors other than 1 and itself. For two given numbers, he wants to know how many prime numbers lie between them, including both the numbers, and which is the largest of those prime numbers.

#include <iostream>
using namespace std;
bool isPrime(int x) {
    if (x < 2) {
        return false;
    }
    for (int i = 2; i * i <= x; i++) {
        if (x % i == 0) {
            return false;
        }
    }
    return true;
}
int main() {
    int a, b;
    cin >> a >> b;
    int count = 0;
    int largest = 0;
    for (int i = a; i <= b; i++) {
        if (isPrime(i)) {
            count = count + 1;
            largest = i;
        }
    }
    cout << "Prime Count: " << count << endl;
    if (count == 0) {
        cout << "Largest Prime: None" << endl;
    } else {
        cout << "Largest Prime: " << largest << endl;
    }
    return 0;
}

#include <iostream>
#include <iomanip>
using namespace std;
class TemperatureLog {
    double readings[10];
    int count;
public:
    TemperatureLog() {
        count = 0;
    }
    void addReading(double t) {
        readings[count] = t;
        count = count + 1;
    }
    double getHighest() {
        double best = readings[0];
        for (int i = 1; i < count; i++) {
            if (readings[i] > best) {
                best = readings[i];
            }
        }
        return best;
    }
    double getLowest() {
        double least = readings[0];
        for (int i = 1; i < count; i++) {
            if (readings[i] < least) {
                least = readings[i];
            }
        }
        return least;
    }
    double getAverage() {
        double sum = 0;
        for (int i = 0; i < count; i++) {
            sum = sum + readings[i];
        }
        return sum / count;
    }
};
int main() {
    int n;
    cin >> n;
    TemperatureLog log;
    for (int i = 0; i < n; i++) {
        double t;
        cin >> t;
        log.addReading(t);
    }
    cout << fixed << setprecision(2);
    cout << "Highest: " << log.getHighest() << endl;
    cout << "Lowest: " << log.getLowest() << endl;

}

#include <iostream>
#include <string>
using namespace std;
class Timer {
    string label;
    int seconds;
public:
    Timer(string label, int seconds) {
        this->label = label;
        this->seconds = seconds;
        cout << "Timer Created: " << label << endl;
    }
    void start() {
        for (int i = seconds; i >= 1; i--) {
            cout << i << endl;
        }
        cout << "Time Up" << endl;
    }
    ~Timer() {
        cout << "Timer Destroyed: " << label << endl;
    }
};
int main() {
    string label;
    int seconds;
    cin >> label >> seconds;
    Timer t(label, seconds);
    t.start();
    return 0;
}
Q4. Problem Statement
Vimala is designing a wall calendar and has to print the number of days in a given month of a given year. For this, she needs to know whether the year is a leap year. A year is a leap year if it is divisible by 400, or if it is divisible by 4 but not divisible by 100. 
In a leap year, February has 29 days, and in other years February has 28 days. The months April, June, September, and November have 30 days, and all the other months have 31 days.
Write a program that reads a year and a month number, and prints whether the year is a leap year and the number of days in the given month.
Input Format 
The first line of input consists of two space-separated integers year and month representing the year and the month number, where 1 is January and 12 is December.
Output Format 
The first line of output prints "Leap Year: Yes" if the year is a leap year, or "Leap Year: No" otherwise. The second line of output prints "Days: " followed by an integer representing the number of days in the given month of the given year.
Refer to the sample output for the formatting specifications.
Constraints 
In this scenario, the test cases fall under the following constraints:
1000 ≤ year ≤ 9999
1 ≤ month ≤ 12

Solution
#include <iostream>
using namespace std;
int main() {
    int year, month;
    cin >> year >> month;
    bool leap = false;
    if (year % 400 == 0) {
        leap = true;
    } else if (year % 100 == 0) {
        leap = false;
    } else if (year % 4 == 0) {
        leap = true;
    }
    int days;
    if (month == 2) {
        if (leap) {
            days = 29;
        } else {
            days = 28;
        }
    } else if (month == 4 || month == 6 || month == 9 || month == 11) {
        days = 30;
    } else {
        days = 31;
    }
    if (leap) {
        cout << "Leap Year: Yes" << endl;
    } else {
        cout << "Leap Year: No" << endl;
    }
    cout << "Days: " << days << endl;
    return 0;
}
